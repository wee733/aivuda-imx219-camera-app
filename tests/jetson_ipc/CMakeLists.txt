cmake_minimum_required(VERSION 3.22)
project(imx219_ipc_smoke LANGUAGES CXX)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cuda_buffer REQUIRED)
find_package(CUDAToolkit REQUIRED)
add_executable(ipc_smoke smoke.cpp)
target_compile_features(ipc_smoke PRIVATE cxx_std_20)
target_link_libraries(ipc_smoke PRIVATE rclcpp::rclcpp ${sensor_msgs_TARGETS} sensor_msgs::sensor_msgs_library cuda_buffer::cuda_buffer CUDA::cudart)
